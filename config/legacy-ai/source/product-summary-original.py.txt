_ALIASES = {alias: category for category, aliases in PRODUCT_GROUPS.items() for alias in aliases}
_SORTED_ALIASES = sorted(_ALIASES, key=len, reverse=True)
_PACKAGING = re.compile(r'(?:套装|套组|组合装|组合套装|礼盒装|礼盒|套件|[一二三四五六七八九十百两\d]+(?:件套|件装|只装|个装|支装))$')
_TAIL_NOTE = re.compile(r'[（(][^（）()]*[）)]$')
_NON_PRODUCT_ENDINGS = ('展示', '演示', '测评', '教程', '摄影', '风格', '造型', '肤色', '皮肤', '店', '工厂', '仓库', '市场')
_NEGATED = re.compile(r'^(?:没有|未见|不含|不是|非商品)')


def _strip_packaging(text):
    previous = None
    while text != previous:
        previous = text
        text = _TAIL_NOTE.sub('', text).strip()
        text = _PACKAGING.sub('', text).strip()
    return text


def _category(text, *, declared=False):
    """以明确商品本体为锚点，任意用途/IP/外观前缀都不能成为新类别。"""
    if not isinstance(text, str) or not text or '#' in text or '＃' in text:
        return None
    text = text.strip()
    if text.endswith(_NON_PRODUCT_ENDINGS) or _NEGATED.search(text):
        return None
    if text in _ALIASES:
        return _ALIASES[text]
    core = _strip_packaging(text)
    # 角色名可以含另一种商品词，以末尾本体为准，例如“面包超人玩具”。
    for alias in _SORTED_ALIASES:
        if core.casefold().endswith(alias.casefold()):
            # 单字商品词需有结构化商品依据，避免把非商品句子当成杯、碗、锅等。
            if len(alias) > 1 or declared or core == alias:
                return _ALIASES[alias]
    if re.search(r'(?:女童|男童|儿童|婴儿|婴幼儿).*?(?:套装|套组|服饰)$', text):
        return '童装'
    if declared and core in {'鱼', '孔雀鱼', '观赏鱼', '斗鱼'}:
        return '鱼'
    if declared and core == '食物':
        return '食品'
    return None


def product_context(items, *, include_issues=False):
    """商品来源只帮助识别商品；无法确定时用None要求保留原标签。"""
    candidates = {}
    for item in items:
        if not isinstance(item, dict):
            continue
        names = [item.get(key) for key in ('name', 'label', 'original_name')]
        category = _category(item.get('category'), declared=True)
        named = {_category(name, declared=True) for name in names} - {None}
        # 历史category可能由旧别名误命中（如吸水杯垫→水杯）。有明确中文本体时先采用本体。
        if item.get('category_source') == 'legacy_name' and named:
            known = next(iter(named)) if len(named) == 1 else _CATEGORY_CONFLICT
        else:
            known = category or (next(iter(named)) if len(named) == 1 else _CATEGORY_CONFLICT if named else None)
        for name in names:
            if isinstance(name, str) and name.strip() and name != '未识别中文品类':
                candidates.setdefault(name.strip(), set()).add(known)
    resolved, issues = {}, {}
    for name, values in candidates.items():
        known = values - {None}
        if len(known) == 1 and _CATEGORY_CONFLICT not in known:
            # 缺字段的记录不能覆盖其他记录中已确认的大类。
            resolved[name] = next(iter(known))
        else:
            resolved[name] = None
            issues[name] = ('多条商品记录的大类不一致，需核验' if known
                            else '已识别为商品，但本地大类映射未覆盖名称或品类')
    return (resolved, issues) if include_issues else resolved


def product_type(tag, womens_context=False, known_products=None):
    if '#' in tag or '＃' in tag:
        return tag
    known_products = known_products or {}
    # 缺失或冲突的商品依据必须保留原词，不能再被名称匹配或女装上下文覆盖。
    if tag in known_products and known_products[tag] is None:
        return tag
    value = known_products.get(tag) or _category(tag)
    if value is None:
        return tag
    if womens_context and value in WOMENS_GARMENTS:
        return '女装'
    return value


def result_products(result):
    labels = result.get('labels') or {}
    attributes = labels.get('attributes') or {}
    products = attributes.get('products') or []
    return [item for item in products if isinstance(item, dict)]
