"""Display product sizes with their recorded product; keep source values intact."""
import re
import unicodedata


_WORDS = {'small': '小号', 'medium': '中号', 'large': '大号',
          'extra small': '加小号', 'extra large': '加大号',
          'one size': '均码', 'onesize': '均码'}
_PRODUCTS = ('连体塑身衣', '塑身衣', '运动内衣', '文胸', '胸罩', '内衣', '连衣裙', '半身裙',
             '卫衣', '衬衫', '毛衣', '上衣', '外套', '背带裤', '裤子', '短裤', '裙子', '手套',
             'T恤', '服装', '衣服', '女装', '男装', '童装', '运动鞋', '鞋子', '靴子', '袜子', '帽子')
_UNKNOWN = {'未知', '无法确定', '无法判断', '无法识别', '不适用', '未注明', '未识别中文品类', '商品', '产品', '其他'}


def size_text(value):
    if not isinstance(value, str):
        return None
    value = unicodedata.normalize('NFKC', value).strip().lstrip('#').strip()
    token = re.sub(r'[-_\s]+', ' ', value).casefold()
    token = re.sub(r'^(?:size|尺码)\s*[:：]?\s*', '', token)
    if token in _WORDS:
        return _WORDS[token]
    code = re.sub(r'\s+', '', token).removesuffix('码')
    if re.fullmatch(r'(?:[xsml]|x{1,4}[sl]|[2-6]x[sl])', code, re.I):
        return code.upper() + '码'
    if re.fullmatch(r'\d{2,3}[a-j]{1,2}', code, re.I):
        return code.upper() + '码'
    if re.fullmatch(r'(?:加小|小|中|大|加大|特大)号|均码|\d{1,3}码', value):
        return value
    return None


def is_bare_size(tag):
    return size_text(tag) is not None


def product_name(item):
    for field in ('name', 'category'):
        value = item.get(field)
        if not isinstance(value, str) or value in _UNKNOWN or not re.search(r'[\u3400-\u9fff]', value):
            continue
        value = unicodedata.normalize('NFKC', value).strip()
        for product in _PRODUCTS:
            if value.casefold().endswith(product.casefold()):
                return '衣服' if product == '服装' else product
        if not re.search(r'[A-Za-z#＃]', value):
            return value
    return None


def product_size_tags(items):
    tags = []
    for item in items:
        if not isinstance(item, dict) or not (name := product_name(item)):
            continue
        sizes = item.get('sizes')
        for size in sizes if isinstance(sizes, list) else []:
            if isinstance(size, dict) and (value := size_text(size.get('value'))):
                tag = name + value
                if tag not in tags:
                    tags.append(tag)
    return tags


def qualify_size_tags(tags, items):
    return list(dict.fromkeys([tag for tag in tags if not is_bare_size(tag)] + product_size_tags(items)))
